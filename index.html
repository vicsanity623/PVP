Server start / stop commands 
# start backend (background)
cd giftquest/backend
nohup venv/bin/python app.py > giftquest.log 2>&1 &

# stop backend
kill $(lsof -t -iTCP:8989 -sTCP:LISTEN)

# public funnel  https://vics-imac-1.tail37b4f2.ts.net:8443
tailscale funnel --bg --yes --https=8443 http://127.0.0.1:8989   # start
tailscale funnel --https=8443 off                                # stop
